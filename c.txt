🧮 3️⃣ Pip Mapping (Deterministic)
// logic/pipMapper.ts

export function mapPipRange(
  volatilityScore: number,
  isJPY: boolean
) {
  const scale = isJPY ? 0.7 : 1;

  const ranges = {
    1: [5, 12],
    2: [10, 25],
    3: [25, 45],
    4: [45, 80],
    5: [80, 150]
  };

  const [min, max] = ranges[volatilityScore as keyof typeof ranges];

  return {
    min: Math.round(min * scale),
    max: Math.round(max * scale)
  };
}

🧠 4️⃣ Memory Adjustment Logic
// logic/memoryAdjuster.ts

export function adjustFromMemory(
  baseScore: number,
  memory: MemoryRecord[]
): number {
  if (memory.length < 3) return baseScore;

  const avgRealized =
    memory.reduce((s, m) => s + (m.realizedPipMove || 0), 0) / memory.length;

  if (avgRealized > 70) return Math.min(5, baseScore + 1);
  if (avgRealized < 15) return Math.max(1, baseScore - 1);

  return baseScore;
}

🌍 5️⃣ Regime Influence
// logic/regimeAdjuster.ts

export function applyRegimeBias(
  score: number,
  regime: MarketRegime
): number {
  if (regime === "Liquidity-Stressed") return Math.min(5, score + 1);
  if (regime === "Pause") return Math.max(1, score - 1);
  return score;
}

🔗 7️⃣ Orchestrator (MAIN PIPELINE)
// services/volatilityEngine.ts

export async function runVolatilityEngine(event: any, regime: MarketRegime) {
  const eventKey = `${event.eventName}_${event.currency}`;

  const memory = await getEventMemory(eventKey, regime);

  let { volatilityScore, volatilityWindow, drivers } =
    await inferVolatility(event);

  volatilityScore = adjustFromMemory(volatilityScore, memory);
  volatilityScore = applyRegimeBias(volatilityScore, regime);

  const expectedPipRange = mapPipRange(
    volatilityScore,
    event.currency === "JPY"
  );

  const confidenceScore = Math.min(
    90,
    50 + memory.length * 2 + volatilityScore * 5
  );

  const result: VolatilityResult = {
    eventKey,
    volatilityScore,
    volatilityWindow,
    expectedPipRange,
    directionalBias: "Two-Way",
    confidenceScore,
    drivers,
    executionNotes:
      volatilityScore >= 4
        ? "Wait for spike then trade continuation"
        : "Avoid event; trade structure"
  };

  return result;
}
